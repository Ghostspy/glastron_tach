cmake_minimum_required(VERSION 3.16)
project(boat_tach_host C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
add_executable(test-model test_model.c ../main/tach_model.c)
target_include_directories(test-model PRIVATE ../main)
target_compile_options(test-model PRIVATE -Wall -Wextra -Werror -pedantic)
target_link_libraries(test-model PRIVATE m)
add_test(NAME tach-model COMMAND test-model)

file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS ../components/lvgl/src/*.c)
add_library(lvgl-host STATIC ${LVGL_SOURCES})
target_include_directories(lvgl-host PUBLIC ../components/lvgl ..)
target_compile_definitions(lvgl-host PUBLIC LV_CONF_INCLUDE_SIMPLE)
target_compile_options(lvgl-host PRIVATE -Wno-unused-parameter)
add_executable(render-ui render_ui.c ../main/tach_ui.c ../main/tach_model.c ../main/tach_font_rpm.c
    ../main/tach_startup.c ../main/glastron_logo.c)
target_include_directories(render-ui PRIVATE ../main)
target_compile_options(render-ui PRIVATE -Wall -Wextra -Werror)
target_link_libraries(render-ui PRIVATE lvgl-host m)

add_executable(test-startup test_startup.c ../main/tach_startup.c ../main/glastron_logo.c)
target_include_directories(test-startup PRIVATE ../main)
target_compile_options(test-startup PRIVATE -Wall -Wextra -Werror)
target_link_libraries(test-startup PRIVATE lvgl-host m)
add_test(NAME startup-timing COMMAND test-startup)

foreach(SCENE normal low high missing rpm-missing voltage-missing stopped night startup)
    add_test(NAME render-${SCENE} COMMAND render-ui ${SCENE} ${CMAKE_CURRENT_BINARY_DIR}/${SCENE}.ppm)
endforeach()
